add_definitions(-DTRANSLATION_DOMAIN=\"krdpserver\")

add_executable(krdpserver)
add_executable(krdp-virtual-guardian virtualguardian.cpp VirtualSessionGuardian.cpp)
target_link_libraries(krdp-virtual-guardian PRIVATE Qt6::Core Qt6::Network)
add_executable(krdp-console-host consolehost.cpp ConsoleSeat.cpp ConsoleWorkerEndpoint.cpp ConsoleWorkerSession.cpp ConsoleHostController.cpp ConsoleWorkerLauncher.cpp)
add_executable(krdp-console-probe consoleprobe.cpp ConsoleSeat.cpp)
add_executable(krdp-greeter-probe greeterprobe.cpp)
add_executable(krdp-audio-route-probe audioprobe.cpp ../src/PipeWireAudioPlayback.cpp)
if (BUILD_PLASMA_SESSION)
    add_executable(krdp-console-worker consoleworker.cpp ConsoleResizeExecutor.cpp ConsoleResizeSession.cpp ConsoleMicrophoneSession.h ../src/PipeWireMicrophone.cpp ../src/PipeWireAudioPlayback.cpp)
    add_executable(krdp-virtual-worker-probe virtualworkerprobe.cpp ConsoleWorkerEndpoint.cpp)
    target_link_libraries(krdp-virtual-worker-probe PRIVATE Qt6::Core Qt6::Gui Qt6::Network KRdp)
    add_executable(krdp-virtual-session-probe virtualsessionprobe.cpp VirtualSessionSupervisor.cpp VirtualSessionControl.cpp ConsoleWorkerEndpoint.cpp)
    target_link_libraries(krdp-virtual-session-probe PRIVATE Qt6::Core Qt6::Gui Qt6::Network KRdp)
    add_executable(krdp-virtual-rdp-host-probe virtualrdphostprobe.cpp VirtualSessionHostController.cpp
        VirtualSessionTransport.cpp VirtualSessionControl.cpp VirtualSessionSupervisor.cpp ConsoleWorkerEndpoint.cpp ConsoleWorkerSession.cpp)
    target_link_libraries(krdp-virtual-rdp-host-probe PRIVATE Qt6::Core Qt6::Gui Qt6::Network KRdp)
    configure_file(org.kde.krdpvirtualprobe.desktop.cmake ${CMAKE_CURRENT_BINARY_DIR}/org.kde.krdpvirtualprobe.desktop @ONLY)
endif()

target_sources(krdpserver PRIVATE main.cpp SessionController.cpp DisplayWakeGuard.cpp PhysicalOutputGuard.cpp HostLayoutExecutor.cpp ConsoleSeat.cpp ConsoleWorkerEndpoint.cpp ConsoleWorkerSession.cpp ConsoleHostController.cpp ConsoleWorkerLauncher.cpp)

kconfig_target_kcfg_file(krdpserver FILE krdpserversettings.kcfg CLASS_NAME ServerConfig SINGLETON)

target_include_directories(krdpserver PRIVATE ${CMAKE_BINARY_DIR})
target_include_directories(krdp-audio-route-probe PRIVATE ${CMAKE_SOURCE_DIR}/src)

target_link_libraries(krdpserver PRIVATE Qt6::Gui Qt6::Network Qt6::DBus KF6::CoreAddons KF6::ConfigGui KF6::DBusAddons KF6::Crash KRdp qt6keychain KF6::StatusNotifierItem KF6::I18n KF6::GlobalAccel)
target_link_libraries(krdp-console-host PRIVATE Qt6::Core Qt6::Gui Qt6::Network Qt6::DBus KRdp)
target_link_libraries(krdp-console-probe PRIVATE Qt6::Core Qt6::DBus)
target_link_libraries(krdp-greeter-probe PRIVATE Qt6::Core Qt6::Gui)
target_link_libraries(krdp-audio-route-probe PRIVATE Qt6::Core PkgConfig::PipeWire)
if (BUILD_PLASMA_SESSION)
    target_link_libraries(krdp-console-worker PRIVATE Qt6::Gui Qt6::Network Qt6::DBus KF6::GlobalAccel KRdp PkgConfig::PipeWire)
endif()

if (BUILD_PLASMA_SESSION)
    target_compile_definitions(krdpserver PRIVATE -DWITH_PLASMA_SESSION)
endif()

configure_file(org.kde.krdpserver.desktop.cmake ${CMAKE_CURRENT_BINARY_DIR}/org.kde.krdpserver.desktop @ONLY)
configure_file(krdp-console-host.service.in ${CMAKE_CURRENT_BINARY_DIR}/krdp-console-host.service @ONLY)
if (BUILD_PLASMA_SESSION)
    configure_file(org.kde.krdpconsoleworker.desktop.cmake ${CMAKE_CURRENT_BINARY_DIR}/org.kde.krdpconsoleworker.desktop @ONLY)
endif()
configure_file(app-org.kde.krdpserver.service.in ${CMAKE_CURRENT_BINARY_DIR}/app-org.kde.krdpserver.service @ONLY)

install(TARGETS krdpserver DESTINATION ${KDE_INSTALL_BINDIR})
install(TARGETS krdp-console-host DESTINATION ${KDE_INSTALL_BINDIR})
install(TARGETS krdp-console-probe DESTINATION ${KDE_INSTALL_BINDIR})
install(TARGETS krdp-greeter-probe DESTINATION ${KDE_INSTALL_BINDIR})
install(TARGETS krdp-audio-route-probe DESTINATION ${KDE_INSTALL_BINDIR})
if (BUILD_PLASMA_SESSION)
    install(TARGETS krdp-console-worker DESTINATION ${KDE_INSTALL_BINDIR})
endif()
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/org.kde.krdpserver.desktop DESTINATION ${KDE_INSTALL_APPDIR})
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/krdp-console-host.service DESTINATION ${KDE_INSTALL_SYSTEMDUNITDIR})
install(FILES console-host.conf.example DESTINATION ${KDE_INSTALL_DATADIR}/krdp)
if (BUILD_PLASMA_SESSION)
    install(FILES ${CMAKE_CURRENT_BINARY_DIR}/org.kde.krdpconsoleworker.desktop DESTINATION ${KDE_INSTALL_APPDIR})
endif()
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/app-org.kde.krdpserver.service DESTINATION ${KDE_INSTALL_SYSTEMDUSERUNITDIR})
install(FILES 00-krdp.preset DESTINATION ${KDE_INSTALL_SYSTEMDUNITDIR}/user-preset)
