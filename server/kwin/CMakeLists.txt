cmake_minimum_required(VERSION 3.25)
project(FarsideKWinPointerCapture LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.9 CONFIG REQUIRED Core DBus Gui Quick Widgets)
find_package(KF6Config CONFIG REQUIRED)
find_package(KF6CoreAddons CONFIG REQUIRED)
find_package(KF6WindowSystem CONFIG REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(Wayland REQUIRED IMPORTED_TARGET wayland-server)
pkg_check_modules(Epoxy REQUIRED IMPORTED_TARGET epoxy)
pkg_check_modules(Drm REQUIRED IMPORTED_TARGET libdrm)
if(NOT EXISTS "${FARSIDE_KWIN_INCLUDE_DIR}/plugin.h")
    message(FATAL_ERROR "Set FARSIDE_KWIN_INCLUDE_DIR to the exact-version KWin SDK")
endif()
file(READ "${FARSIDE_KWIN_INCLUDE_DIR}/config-kwin.h" farside_kwin_config)
if(NOT farside_kwin_config MATCHES "KWIN_PLUGIN_VERSION_STRING \"6\\.6\\.6\"")
    message(FATAL_ERROR "The native bridge currently supports the exact KWin 6.6.6 ABI only")
endif()
find_library(FARSIDE_KWIN_LIBRARY NAMES kwin libkwin.so.6 REQUIRED)
add_library(farside-pointer-capture MODULE FarsidePointerCapture.cpp)
set_target_properties(farside-pointer-capture PROPERTIES PREFIX "")
target_include_directories(farside-pointer-capture PRIVATE ${FARSIDE_KWIN_INCLUDE_DIR} ${CMAKE_CURRENT_SOURCE_DIR}/upstream)
target_link_libraries(farside-pointer-capture PRIVATE ${FARSIDE_KWIN_LIBRARY} Qt6::Core Qt6::DBus Qt6::Gui Qt6::Quick Qt6::Widgets KF6::ConfigCore KF6::CoreAddons KF6::WindowSystem PkgConfig::Wayland PkgConfig::Epoxy PkgConfig::Drm)
include(GNUInstallDirs)
install(TARGETS farside-pointer-capture LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}/qt6/plugins/kwin/plugins)
option(FARSIDE_BUILD_NATIVE_CAPTURE_TEST "Build the explicitly selected isolated compositor test" OFF)
if(FARSIDE_BUILD_NATIVE_CAPTURE_TEST)
    find_package(Qt6 CONFIG REQUIRED Test GuiPrivate WaylandClient)
    add_executable(NativePointerCaptureTest NativePointerCaptureTest.cpp ../KWinPointerCapture.cpp)
    target_include_directories(NativePointerCaptureTest PRIVATE .. ../../src)
    target_link_libraries(NativePointerCaptureTest PRIVATE Qt6::Test Qt6::Gui Qt6::GuiPrivate Qt6::Quick Qt6::DBus Qt6::WaylandClient)
    qt6_generate_wayland_protocol_client_sources(NativePointerCaptureTest FILES
        /usr/share/plasma-wayland-protocols/fake-input.xml
        ${CMAKE_CURRENT_SOURCE_DIR}/protocols/pointer-constraints-unstable-v1.xml)
endif()
